File 1 — src/pages/essays/[slug].astro (dynamic essay route):

  ---
  /**
   * /essays/[slug] — dynamic essay route.
   *
   * D-28 draft filter applied INSIDE getStaticPaths — drafts cannot reach
   * the build output, the sitemap, or RSS (RSS is implemented in 02-05).
   *
   * Reading-time computed inline from entry.body word count (CD-09):
   *   - Skipped if word count < 500 (CD-09 lower bound).
   *   - 250 wpm baseline, ceiling.
   * Inline computation chosen over a remark plugin to keep the dependency
   * surface small for Phase 2.
   */
  import EssayLayout from '../../layouts/EssayLayout.astro';
  import { getCollection, render } from 'astro:content';
  import type { GetStaticPaths } from 'astro';

  export const getStaticPaths = (async () => {
    const essays = await getCollection('essays', ({ data }) => !data.draft);
    return essays.map((entry) => ({
      params: { slug: entry.id.replace(/\.md$/, '') },
      props: { entry },
    }));
  }) satisfies GetStaticPaths;

  const { entry } = Astro.props;
  const { Content } = await render(entry);

  // CD-09: inline reading-time. Body word count from entry.body; ceiling at
  // 250 wpm; hide if < 500 words. EssayLayout treats `undefined` or values
  // < 1 as "do not display."
  const wordCount = (entry.body ?? '').trim().split(/\s+/).filter(Boolean).length;
  const readingMinutes = wordCount >= 500 ? Math.max(1, Math.ceil(wordCount / 250)) : undefined;
  ---
  <EssayLayout entry={entry} readingMinutes={readingMinutes}>
    <Content />
  </EssayLayout>

  import BaseLayout from '../../layouts/BaseLayout.astro';
  import { getAllTags, getEntriesByTag } from '../../lib/tags';
  import type { GetStaticPaths } from 'astro';

  export const getStaticPaths = (async () => {
    const tags = await getAllTags();
    return tags.map((tag) => ({
      params: { tag },
      props: { tag },
    }));
  }) satisfies GetStaticPaths;

  const { tag } = Astro.props;
  const entries = await getEntriesByTag(tag);

  const title = `Topic: ${tag}`;
  const description = `All essays and notes tagged ${tag}.`;
  const canonical = `/topics/${tag}`;

  const hrefFor = (entry: typeof entries[number]) => {
    const slug = entry.id.replace(/\.md$/, '');
    return entry.collection === 'essays' ? `/essays/${slug}` : `/notes/${slug}`;
  };
  ---
  <BaseLayout
    title={title}
    description={description}
    canonical={canonical}
    jsonLdSchema="webpage"
    jsonLdData={{ name: title, url: new URL(canonical, Astro.site).toString(), description }}
  >
    <article class="topic h-feed">
      <header class="topic__header">
        <p class="topic__label">Topic</p>
        <h1 class="topic__h1 p-name">{tag}</h1>
      </header>
      <ul class="topic__list">
        {entries.map((e) => (
          <li class="h-entry">
            <a href={hrefFor(e)} class="topic__link">
              <span class="topic__collection">{e.collection === 'essays' ? 'Essay' : 'Note'}</span>
              <span class="topic__title p-name">{e.data.title}</span>
              {e.data.subtitle && <span class="topic__subtitle p-summary">{e.data.subtitle}</span>}
            </a>
          </li>
        ))}
      </ul>
    </article>
  </BaseLayout>
