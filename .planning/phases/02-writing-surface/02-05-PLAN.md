File 1 — src/pages/rss.xml.ts (combined feed):

  /**
   * /rss.xml — combined feed (essays + notes), full content per CD-05.
   *
   * Per .planning/research/PITFALLS.md §11 (RSS broken):
   *   - Use the official @astrojs/rss helper (NOT roll-your-own XML).
   *   - Validate at https://validator.w3.org/feed/ before launch (Wave 4 audit).
   *   - Test in NetNewsWire / Reeder before public link.
   *
   * D-28 drafts excluded via getCollection filter callback.
   * Sort key: updated ?? published, descending.
   *
   * CD-05: full body content in <description>. Peer audience reads in feed
   * readers; full content removes the click-through nag.
   */
  import rss from '@astrojs/rss';
  import { getCollection } from 'astro:content';
  import type { APIContext } from 'astro';
  import { marked } from 'marked';

  export async function GET(context: APIContext) {
    if (!context.site) {
      throw new Error('Astro.site is undefined — set PUBLIC_SITE_URL in astro.config.mjs');
    }
    const [essays, notes] = await Promise.all([
      getCollection('essays', ({ data }) => !data.draft),
      getCollection('notes', ({ data }) => !data.draft),
    ]);
    const all = [...essays, ...notes].sort(
      (a, b) => (b.data.updated ?? b.data.published).valueOf() - (a.data.updated ?? a.data.published).valueOf()
    );
    return rss({
      title: 'Wesley Schlemmer — Writing',
      description: 'Essays and notes — Freedom Tech, Bitcoin, sovereign AI, opting out of legacy systems.',
      site: context.site,
      items: await Promise.all(all.map(async (entry) => {
        const slug = entry.id.replace(/\.md$/, '');
        const link = entry.collection === 'essays' ? `/essays/${slug}` : `/notes/${slug}`;
        return {
          title: entry.data.title,
          pubDate: entry.data.published,
          description: entry.data.description,
          link,
          content: await marked(entry.body ?? ''),
        };
      })),
      customData: '<language>en-us</language>',
    });
  }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Markdown body -> RSS XML | entry.body is markdown (treated as text by RSS spec); @astrojs/rss handles XML escaping of embedded characters |
| Frontmatter title/description -> RSS XML | Same — escaping handled by @astrojs/rss |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-05-01 | Tampering | XML injection via markdown body | mitigate | @astrojs/rss runs every item field through proper XML encoding (CDATA-wrapped or entity-encoded). The plan does NOT roll its own XML or string-concat — it passes structured objects to the helper. PITFALLS §11 explicitly calls out roll-your-own XML as the failure mode. |
| T-02-05-02 | Information Disclosure | Draft leak via feed | mitigate | All three feeds use `getCollection(name, ({ data }) => !data.draft)`. Acceptance criteria asserts the predicate appears in all three files. |
| T-02-05-03 | Information Disclosure | Future-dated post leak | accept | Astro emits the feed at build time; if a post has a future `published` date, it ships with a future pubDate. Wesley owns publish discipline; not Phase 2 scope. |
| T-02-05-04 | Spoofing | Forged RSS GUID | accept | @astrojs/rss derives the GUID from the link by default. A user controlling frontmatter could change the slug, but cannot impersonate someone else's feed. |
| T-02-05-05 | Tampering | Feed-discovery to wrong host | mitigate | `<link rel="alternate">` URLs derive from Astro.site (PUBLIC_SITE_URL), not from a request header — staging vs production host is correct by config, not by request. |
| T-02-05-06 | Information Disclosure | Markdown body containing private references | accept | Wesley owns content; if a draft accidentally references private data, the D-28 filter prevents publish. The two-step draft pipeline (Wave 3 plans 02-07/02-08) requires explicit `draft: false` flip on approval. |
</threat_model>
