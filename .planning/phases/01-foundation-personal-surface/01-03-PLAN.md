    ```astro
    ---
    import type { Person, WebSite, WebPage, BreadcrumbList } from 'schema-dts';

    interface Props {
      schema: 'website' | 'person' | 'webpage' | 'breadcrumb';
      data?: Record<string, unknown>;
    }

    const { schema, data = {} } = Astro.props;

    if (!Astro.site) {
      throw new Error('Astro.site is undefined — set PUBLIC_SITE_URL in astro.config.mjs');
    }
    const siteUrl = Astro.site.toString().replace(/\/$/, '');
    // Person @id is pinned to the canonical apex (intentional graph-stable identity per RESEARCH §JSON-LD line 433).
    // This is the ONLY hardcoded crossthebridge.io reference allowed in the component (Pitfall G exception).
    const PERSON_ID = 'https://crossthebridge.io/about#wesley';

    const PERSON: Person = {
      '@type': 'Person',
      '@id': PERSON_ID,
      name: 'Wesley Pyburn',
      jobTitle: 'Founder, Cross The Bridge',
      url: `${siteUrl}/about`,
      image: `${siteUrl}/wesley-headshot.jpg`,
      sameAs: [],   // populate when Wesley supplies public profile URLs
    };

    const WEBSITE: WebSite = {
      '@type': 'WebSite',
      '@id': `${siteUrl}/#website`,
      name: 'Cross The Bridge',
      url: `${siteUrl}/`,
      description: "Wesley Pyburn's personal site — Freedom Tech, Bitcoin, sovereign AI, and the work of opting out of legacy systems.",
      publisher: { '@id': PERSON_ID } as any,
      inLanguage: 'en',
    };

    let json: Record<string, unknown>;
    if (schema === 'website') {
      json = { '@context': 'https://schema.org', ...WEBSITE };
    } else if (schema === 'person') {
      json = { '@context': 'https://schema.org', ...PERSON };
    } else if (schema === 'breadcrumb') {
      // Caller supplies `data.breadcrumb` array AND optionally `data.webpage` for the WebPage half.
      // The output is two scripts so consumers can pass both or just one.
      json = { '@context': 'https://schema.org', '@type': 'BreadcrumbList', ...data };
    } else {
      // schema === 'webpage'
      const fallback: WebPage = {
        '@type': 'WebPage',
        url: data.url as string ?? siteUrl,
        name: data.name as string ?? 'Cross The Bridge',
        isPartOf: { '@id': `${siteUrl}/#website` } as any,
      };
      json = { '@context': 'https://schema.org', ...fallback, ...data };
    }
    ---
    <script type="application/ld+json" set:html={JSON.stringify(json)} />
    ```

    **Notes on type-safety:** `schema-dts` makes `publisher: { '@id': ... }` and `isPartOf: { '@id': ... }` strict — they expect a full Organization/WebSite object. Cast via `as any` is the documented escape hatch when you only want to emit the `@id` reference (RESEARCH.md line 442 example uses the same pattern). This still type-checks the rest of the schema; it just allows the `@id`-only ref form.

    **Notes on per-route consumption (consumed by PLAN-07/08/09):**
    - Homepage (`/`): `<JsonLd schema="website" />`
    - About (`/about`): `<JsonLd schema="person" />`
    - Project pages (`/projects/*`): `<JsonLd schema="breadcrumb" data={{ itemListElement: [...] }} />` — caller supplies the `BreadcrumbList` items array
    - Contact, Colophon: `<JsonLd schema="webpage" data={{ name: 'Contact', url: '...', description: '...' }} />`

    **DO NOT** hardcode `staging.crossthebridge.io` anywhere in either component (Pitfall G). The ONLY exception is the Person `@id` constant `https://crossthebridge.io/about#wesley` — that's an intentional graph-stable identity that pins to the canonical apex regardless of staging host.
