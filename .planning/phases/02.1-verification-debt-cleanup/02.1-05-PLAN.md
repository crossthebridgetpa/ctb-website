    ```ts
    // WR-03: validate PUBLIC_UMAMI_HOST at build time.
    //   - Unset (undefined or '') -> skip Umami silently (analytics is optional).
    //   - Set + malformed -> throw at build (per locked decision: "no silent failures").
    //   - Set + valid -> strip trailing slashes so `${host}/script.js` is well-formed.
    const rawUmamiHost = import.meta.env.PUBLIC_UMAMI_HOST;
    let umamiHost: string | undefined;
    if (rawUmamiHost && rawUmamiHost.length > 0) {
      try {
        new URL(rawUmamiHost);
      } catch {
        throw new Error(
          `BaseLayout: PUBLIC_UMAMI_HOST is set but not a valid URL: "${rawUmamiHost}". ` +
            `Set to an https URL like https://umami.example.com, or unset to disable analytics.`
        );
      }
      umamiHost = rawUmamiHost.replace(/\/+$/, '');
    }
    const umamiId = import.meta.env.PUBLIC_UMAMI_WEBSITE_ID;
    ```
