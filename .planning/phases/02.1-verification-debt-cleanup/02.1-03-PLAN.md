    **Step 1 — Add a validation helper function** in the frontmatter, after line 31 (`const siteUrl = ...`) and before line 33 (the PERSON_ID block). Insert:

    ```ts
    /**
     * CR-03: fail loud at build time when required Schema.org fields are missing
     * or empty. Per Wesley's locked decision in 02.1-CONTEXT.md: "I don't want
     * things to fail silently." Bad frontmatter fails astro build / Vercel deploy
     * instead of shipping corrupt JSON-LD.
     */
    function requireFields(
      schemaName: string,
      data: Record<string, unknown>,
      fields: readonly string[]
    ): void {
      for (const f of fields) {
        const v = data[f];
        if (v === undefined || v === null || v === '') {
          throw new Error(
            `JsonLd: required ${schemaName} field "${f}" is missing or empty. ` +
              `Caller must supply this in the data prop. ` +
              `Page: ${Astro.url?.pathname ?? '(unknown)'}.`
          );
        }
      }
    }
    ```
